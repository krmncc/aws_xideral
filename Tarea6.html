# Gráfica comparativa de 4 caracteristicas musicales

columnas = ['speechiness_%', 'liveness_%', 'instrumentalness_%', 'acousticness_%']
promedios = df[columnas].mean()

colores = ['#4C72B0', '#55A868', '#C44E52', '#8172B2']
barras = plt.bar(promedios.index, promedios.values, color=colores, edgecolor='black', alpha=0.85)
plt.title('Comparativa de Promedios de las 4 Características', fontsize=14, fontweight='bold', pad=15)
plt.xlabel('Características Analizadas', fontsize=12, labelpad=10)
plt.ylabel('Valor Promedio General', fontsize=12, labelpad=10)
plt.grid(axis='y', linestyle='--', alpha=0.5)

for barra in barras:
    alto = barra.get_height()
    plt.text(
        barra.get_x() + barra.get_width()/2,  # Posición X (centro de la barra)
        alto + (alto * 0.01),                 # Posición Y (ligeramente arriba del borde)
        f'{alto:.2f}',                        # Formato con 2 decimales
        ha='center', va='bottom', fontsize=11, fontweight='bold'
    )

plt.tight_layout()
plt.show()
